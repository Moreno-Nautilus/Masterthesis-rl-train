# Working dependency pins for the `serl` conda env (CPU-only), verified 2026-08-26.
#
# SERL/serl_launcher was written against the JAX 0.4.x ecosystem; the upstream
# requirements.txt uses `>=` pins that now resolve to JAX 0.6 / flax 0.10 / TF 2.21,
# which break serl_launcher (jax.tree_map removed in 0.6; ml_dtypes/protobuf clashes).
# These pins are the compatible window that makes the full async DrQ+RLPD loop run.
#
# Setup:
#   conda create -n serl python=3.10
#   conda activate serl
#   pip install -e serl_launcher            # pulls agentlace (pinned git commit)
#   pip install -r serl_launcher/requirements.txt
#   pip install -e iiwa_serl
#   pip install -r iiwa_serl/requirements-serl-cpu.txt   # <-- this file (pins fix)
#   pip install gymnasium pygame
#
# GPU note: keep CPU-only while the PPO training/deploy is live (no GPU contention).
# For real training later, swap jax to: pip install "jax[cuda12_pip]==0.4.35" -f \
#   https://storage.googleapis.com/jax-releases/jax_cuda_releases.html
# (self-contained CUDA wheels, do NOT modify the system NVIDIA driver.)
